cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
project(NextGenStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
option(STUDIO_DEPLOY_RUNTIME "Bundle Qt runtime during install" OFF)
option(STUDIO_ENABLE_IPO "Enable link-time optimization (measure before enabling)" OFF)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
qt_standard_project_setup()
if(STUDIO_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(NOT ipo_supported)
    message(FATAL_ERROR "IPO unavailable: ${ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
add_library(studio_document src/document.cpp src/document.h)
target_link_libraries(studio_document PUBLIC Qt6::Core)
target_include_directories(studio_document PUBLIC src)
qt_add_executable(nextgen-studio src/main.cpp src/mainwindow.cpp src/mainwindow.h src/canvas.cpp src/canvas.h)
target_link_libraries(nextgen-studio PRIVATE studio_document Qt6::Widgets)
qt_add_resources(nextgen-studio studio_resources PREFIX "/" FILES resources/theme.qss examples/welcome.otui integration/dev_studio_bridge/bridge.lua integration/dev_studio_bridge/dev_studio_bridge.otmod)
set_target_properties(nextgen-studio PROPERTIES WIN32_EXECUTABLE TRUE)
foreach(target studio_document nextgen-studio)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O2>)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:-O2>)
  endif()
endforeach()
if(BUILD_TESTING)
  qt_add_executable(document-tests tests/document_tests.cpp)
  target_link_libraries(document-tests PRIVATE studio_document)
  if(MSVC)
    target_compile_options(document-tests PRIVATE /utf-8)
  endif()
  add_test(NAME document-integrity COMMAND document-tests)
  add_test(NAME desktop-smoke COMMAND nextgen-studio --smoke-test)
  set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
install(TARGETS nextgen-studio RUNTIME DESTINATION bin)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
install(DIRECTORY examples integration DESTINATION share/nextgen-studio)
if(WIN32)
  install(FILES $<TARGET_PDB_FILE:nextgen-studio> DESTINATION bin OPTIONAL)
endif()
if(STUDIO_DEPLOY_RUNTIME)
  qt_generate_deploy_app_script(TARGET nextgen-studio OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
set(CPACK_PACKAGE_NAME NextGen-Studio)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
